## 前情提要
这一节用 Score + Student + std::unordered_map 搭一个命令行成绩管理小程序，支持增删改查、按总分/均分/单科排序、班级统计、文件读写。  
注释版盯住几处易踩的坑：  
get_score 用 -1 表示未找到，因为 0 是合法成绩；  
set_score 必须按引用遍历 Score，否则改的是副本；  
op4 的链式 set_* 调用依赖每个 setter 返回 *this；  
op5 的 classmin 初值不能是 -1，且 classsum / (1.0)*numss 实际是乘法而非除法。

#include "Student.h"

void Student::show_score(){
    using namespace std;
    if(v.size() == 0){
        cout << "no score" << '\n';
        return;
    }
    for(Score x : v){
        cout << x.get_kind() << ": " << x.get_grade() << '\n';
    }
    cout << "===========" << '\n';
    return;
}

double Student::get_score(string kindname) const{
    for(Score x : v){
        if(x.get_kind() == kindname){
            return x.get_grade();
        }
    }
    // 不能改成返回 0 表示未找到，0 是合法成绩，调用方无法区分“没这门课”和“考了 0 分”
    return -1;
}

string Student::get_name(){
    return name;
}

Student& Student::set_num(string newnum) {

}

    return *this;
}

    sex = newsex;
    return *this;

    age = newage;
    return *this;
}

Student& Student::set_cla(string newcla){
    cla = newcla;
    return *this;
}

bool Student::set_score(string newkind, double newgrade){
    using namespace std;
    if(v.size() == 0){
        cout << "nofind" << '\n';
        return false;
    }
    // 不能按值遍历 Score x，修改的是副本，原 vector 里的成绩不变
    for(Score &x : v){
        if(x.get_kind() == newkind){
            x.set_score(newgrade);
            return true;
        }
    }
    cout << "nofind" << '\n';
    return false;
}

Student& Student::add_sco(Score sc) {
    v.push_back(sc);
    return *this;
}

double Student::get_sum() const{
    double sum = 0;
    for(Score x : v){
        sum += x.get_grade();
    }
    return sum;
}

double Student::get_ave() const{
    double sum = get_sum();
    double nums = static_cast<double>(v.size());
    if(nums == 0){
        std::cout << "/ 0" << '\n';
        return 0.0;
    }
    return sum/nums;
}

bool cmpsum(const Student& a, const Student& b){
    return a.get_sum() > b.get_sum();
}

bool cmpave(const Student& a, const Student& b){
    return a.get_ave() > b.get_ave();
}

void Student::fileall(std::fstream& file){
    using namespace std;
    file << "========================" << '\n';
    file << "number: " << num << '\n';
    file << "name: " << name << '\n';
    file << "sex: " << sex << '\n';
    file << "age: " << age << '\n';
    file << "class:" << cla << '\n';
    if(v.size() == 0){
        file << "no score" << '\n';
    }
    else{
        for(Score x : v){
            file << x.get_kind() << ": " << x.get_grade() << '\n';
        }
    }
    file << "summary: " << get_sum() << '\n';
    file << "average" << get_ave() << '\n';
    file << "========================" << '\n';
}

#include "Student.h"

    int choice;

void op1(){
    using namespace std;
    while(1){
        cout << "========================" << '\n';
        cout << "[a]just a student" << '\n';
        cout << "[b]show all with summary" << '\n';
        cout << "[c]show all with average" << '\n';
        cout << "[d]show all with subject" << '\n';
        cout << "[q]exit" << '\n';
        char choice;
        cin >> choice;
        if(choice == 'q') break;
        if(choice == 'a'){
            cout << "cin number and name" << '\n';
            string nums, nams;
            cin >> nums >> nams;    
            auto it = maps.find(nums);
            if(it == maps.end()){
                cout << "no find" << '\n';
                continue;
            }
            if(it->second.get_name() != nams){
                cout << "name is error" << '\n';
                continue;
            }
            it->second.showall();
        }
        if(choice == 'b'){
            vector<Student> vs;
            for(auto it : maps){
                vs.push_back(it.second);
            }
            sort(vs.begin(), vs.end(), cmpsum);
            for(auto x : vs){
                x.showall();
            }
        }
        if(choice == 'c'){
            vector<Student> vs;
            for(auto it : maps){
                vs.push_back(it.second);
            }
            sort(vs.begin(), vs.end(), cmpave);
            for(auto x : vs){
                x.showall();
            }
        }
        if(choice == 'd'){
            cout << "cin the subject" << '\n';
            string sub;
            cin >> sub;
            vector<Student> vs;
            for(auto it : maps){
                vs.push_back(it.second);
            }
            sort(vs.begin(), vs.end(),[&sub](const Student& a, const Student& b){
                return a.get_score(sub) > b.get_score(sub);
            });
            for(auto x : vs){
                x.showall();
            }
        }
    }
    
}

void op5(){
    using namespace std;
    cout << "========================" << '\n';
    //支持统计班级总分、平均分、最高分、最低分
    double classsum = 0.0, classave, classmax = -1.0, classmin = -1.0;
    int numss = 0;
    for(auto it : maps){
        classsum += it.second.get_sum();
        classmax = max(classmax, it.second.get_sum());
        // 不能把 classmin 初始化为 -1，总分非负，min(-1, sum) 恒为 -1，最小值统计错误
        classmin = min(classmin, it.second.get_sum());
        numss++;
    }
    // 不能写成 classsum / (1.0)*numss，乘除同优先级左结合，实际是 classsum*numss，平均分错误
    classave = classsum / (1.0)*numss;
    cout << "class summary: " << classsum << '\n';
    cout << "class average: " << classave << '\n';
    cout << "class max: " << classmax << '\n';
    cout << "class min: " << classmin << '\n';
    cout << "========================" << '\n';
}
