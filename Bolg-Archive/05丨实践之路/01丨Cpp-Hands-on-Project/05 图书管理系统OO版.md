## 前情提要
这一节做面向对象版的图书管理：  
LibaryItem 抽象基类、Book/Magazine 两个派生类、Storage 抽象存储接口、TextStorage 文本实现，  
Libary 负责增删、借还、排序、统计与文件存取，main.cpp 提供菜单。  
注释版重点两处：  
Libary::add 的参数必须是 unique_ptr& 而非 const&，因为 push_back(std::move(item)) 需要非 const 左值转移所有权；  
TextStorage::load 末尾的 if(!is.bad()) 疑似条件写反了，正常读完反而会误抛异常，需要实际跑一次加载验证。

#include "Libary.h"
#include "LibaryItem.h"
using std::cout;
using std::cin;

// 不能改成 const 引用，push_back(std::move(item)) 需要非 const 左值来转移所有权
void Libary::add(std::unique_ptr<LibaryItem>& item) {
    for (const auto& x : items) {
        if (x->getid() == item->getid()) {
            throw std::invalid_argument("duplicate id: " + item->getid());
        }
    }
    items.push_back(std::move(item));
}

void Libary::detele(std::unique_ptr<LibaryItem>& item) {
    int index = 0;
    for(auto &x : items){
        if(x->getname() == item->getname() 
        && x->getid() == item->getid()
        && x->kind() == item->kind()){
            items.erase(items.begin() + index);
            return;
        }
        index++;
    }
    throw std::out_of_range("no find item to delete");
    return ;
}

void Libary::find(std::string ids) {
    int index = 0;
    for(auto &x : items){
        if(ids == x->getid()){
            x->Show();
            return;
        }
        index++;
    }
    throw std::out_of_range("no result");
    return ;
}

void Libary::list() {
    for(auto &x : items){
        x->Show();
    }
    return;
}

void Libary::borrow(std::unique_ptr<LibaryItem>& item) {
    for(auto &x : items){
        if(x->getname() == item->getname() 
        && x->getid() == item->getid()
        && x->kind() == item->kind()){
            if(x->Check() == true ){
                cout << "borrowed" << '\n';
                return;
            }
            x->setstatus();
            return;
        }
    }
    throw std::out_of_range("no find item to borrow");
    return ;
}

void Libary::returnitem(std::unique_ptr<LibaryItem>& item) {
    for(auto &x : items){
        if(x->getname() == item->getname() 
        && x->getid() == item->getid()
        && x->kind() == item->kind()){
            if(x->Check() == false ){
                cout << "not borrowed" << '\n';
                return;
            }
            x->setstatus();
            return;
        }
    }
    throw std::out_of_range("no find item to return");
    return ;
}

void Libary::sortwithname() {
    std::sort(items.begin(), items.end(),[](std::unique_ptr<LibaryItem>& a, std::unique_ptr<LibaryItem>& b){
        return a->getname() < b->getname();
    });
    return;
}

void Libary::showtotal() {
    int cnt = 0, b = 0, nob = 0;
    for(auto &x : items){
        cnt++;
        if(x->Check()) b++;
        else nob++;
    }
    cout << "===================" << '\n';
    cout << "total    : " << cnt << '\n';
    cout << "borrowed : " << b << '\n';
    cout << "haved    : " << nob << '\n';
    cout << "===================" << '\n';

    return;
}

std::vector<std::unique_ptr<LibaryItem>>& Libary::libstream() {
    return items;
}

void Libary::loadfile() {
    items = sto.load();
}

#include "TextStorage.h"
#include <fstream>
#include <sstream>

void TextStorage::save(std::vector<std::unique_ptr<LibaryItem>>& items) {
    std::string na = "data.txt";
    std::ofstream os(na);
    if(!os.is_open()){
        throw std::runtime_error("file open failed");
    }

    for(auto &x : items){
        os << "=======================" << '\n';
        os << "kind  : " << x->kind() << '\n';
        os << "name  : " << x->getname() << '\n';
        os << "id    : " << x->getid() << '\n';
        os << "status: " << x->Check() << '\n';
        os << "=======================" << '\n';
    }
    
    if(!os){
        throw std::runtime_error("file save failed");
    }

}

std::vector<std::unique_ptr<LibaryItem>> TextStorage::load() {
    using std::string;
    using std::stringstream;
    string na = "data.txt";
    std::ifstream is(na);
    if(!is.is_open()){
        throw std::runtime_error("file open failed");
    }

    std::vector<std::unique_ptr<LibaryItem>> items;
    string line;
    std::size_t linenum;
    int part = 1;
    string nkind, nname, nid, nstatus;

    while(getline(is, line)){
        if(line.empty()) continue;
        if(line == "=======================") continue;
        string res;
        int pos = 0;
        int l = line.size();
        for(int i = 0; i < l; i++){
            if(line[i] == ':'){
                pos = i;
                break;
            }
        }
        pos += 2;
        for(int i = pos; i < l; i++){
            res += line[i];
        }
        if(part == 1){
            nkind = res;
        }
        else if(part == 2){
            nname = res;
        }
        else if(part == 3){
            nid = res;
        }
        else if(part == 4){
            nstatus = res;
            int x = stoi(nstatus);
            bool y = (x == 1 ? true : false);
            part = 0;
            std::unique_ptr<LibaryItem> item;
            if(nkind == "Book"){
                Book tmp(nid, nname, y, nkind);
                item = std::make_unique<Book>(tmp);
            }
            else if(nkind == "Magazine"){
                Magazine tmp(nid, nname, y, nkind);
                item = std::make_unique<Magazine>(tmp);
            }
            else{
                throw std::runtime_error("unknown kind: " + nkind);
            }
            items.push_back(std::move(item));
        }
        part++;
    }
    

    // 疑似反了：正常读完 is.bad() 为 false，!is.bad() 为 true 会误抛异常；需验证读取正常文件是否抛异常
    if(!is.bad()){
        throw std::runtime_error("file load failed");
    }

    return items;
}
